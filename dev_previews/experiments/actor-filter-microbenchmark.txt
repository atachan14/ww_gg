if __name__=='__main__':
 b=initial_board(7,2,1,1)
 states=[fixtures.ActorInformationTests().mixed_state(),TrueScanState(from_board(b),(b.alive,),initial_rule='none',tracked=())]
 for state in states:
  predicate=lambda h:bool(h.roles[0]&1);before=[];after=[]
  assert legacy(state,predicate)==condition_actor_state(state,predicate)
  for _ in range(5):
   before.append(timeit(lambda:legacy(state,predicate),number=100))
   after.append(timeit(lambda:condition_actor_state(state,predicate),number=100))
  print(dict(hypotheses=len(state.hypotheses),before=before,after=after,median_ratio=median(after)/median(before)))
